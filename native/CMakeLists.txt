cmake_minimum_required(VERSION 3.24)
project(BenZicaNative VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build with Visual Studio 2022 x64 and Windows 11 SDK 10.0.26100.0 or newer")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
add_library(BenZicaCameraSource SHARED
  source/dllmain.cpp source/winrtCommon.cpp source/SimpleMediaSource.cpp
  source/SimpleMediaStream.cpp source/SimpleFrameGenerator.cpp source/VirtualCameraMediaSourceActivate.cpp
  source/VirtualCameraMediaSource.def)
target_include_directories(BenZicaCameraSource PRIVATE source vendor/wil/include)
target_compile_options(BenZicaCameraSource PRIVATE /W3 /EHsc /permissive- /Zc:__cplusplus)
target_link_libraries(BenZicaCameraSource PRIVATE mfplat mf mfuuid mfsensorgroup windowsapp advapi32 ole32 propsys)
add_executable(BenZicaCameraHost host/main.cpp)
target_link_libraries(BenZicaCameraHost PRIVATE mfplat mfuuid mfsensorgroup ole32 advapi32)
include(CTest)
if(BUILD_TESTING)
  add_executable(FrameTransportTest tests/FrameTransportTest.cpp)
  target_link_libraries(FrameTransportTest PRIVATE ole32 advapi32)
  add_test(NAME FrameTransportTest COMMAND FrameTransportTest)
endif()
